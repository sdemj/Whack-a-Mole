(Pick the Word 게임에서 완성된 버전을 그대로 뽑아둔 것입니다)
=====================================================================

구성: 왼쪽 위(홈 / 뒤로 / 앞으로) + 오른쪽 위(How to Play / 효과음 / 배경음 / 전체화면)

  [1] 그대로 복사해서 붙여넣으면 되는 것 (HTML + CSS + 효과음/배경음/전체화면 JS)

[1-D] JS - 배경음악 on/off (그대로 사용 가능, <audio id="bgm-intro">/<audio id="bgm-game"> 필요)
=====================================================================
※ 트랙이 2개(인트로용/게임용)가 아니라 다르게 구성된 게임이면 bgmTracks 객체의
   key/개수만 게임 화면 구조에 맞게 바꾸면 됩니다.

const bgmTracks = {
    intro: document.getElementById('bgm-intro'),
    game: document.getElementById('bgm-game')
};
let currentBgmKey = 'intro';
let musicMuted = false; // 초기 상태: 음악 켜짐
let audioUnlocked = false; // 브라우저가 소리 재생을 허용했는지 여부 (사용자 상호작용 필요)

// 잠금 해제 전에는 음소거 상태로 재생(대부분의 브라우저가 음소거 자동재생은 허용함)
function syncBgmAudio(audio) {
    if (!audio) return;
    audio.muted = !audioUnlocked;
}

function tryPlayCurrentBgm() {
    if (musicMuted) return;
    const audio = bgmTracks[currentBgmKey];
    if (!audio) return;
    syncBgmAudio(audio);
    if (audio.paused) audio.play().catch(() => {});
}

// 화면이 바뀔 때 playBgm('intro') / playBgm('game') 처럼 호출해서 트랙 전환
function playBgm(key) {
    if (currentBgmKey === key) return;
    currentBgmKey = key;
    Object.keys(bgmTracks).forEach(k => {
        const audio = bgmTracks[k];
        if (k === key) {
            audio.currentTime = 0;
            if (!musicMuted) {
                syncBgmAudio(audio);
                audio.play().catch(() => {});
            }
        } else {
            audio.pause();
        }
    });
}

function toggleMusic() {
    musicMuted = !musicMuted;
    const audio = bgmTracks[currentBgmKey];
    const icon = document.getElementById('music-toggle-icon');
    if (musicMuted) {
        audio.pause();
        if (icon) icon.className = 'fa-solid fa-volume-xmark';
    } else {
        syncBgmAudio(audio);
        audio.play().catch(() => {});
        if (icon) icon.className = 'fa-solid fa-music';
    }
}

// 페이지 로드 시 음소거 상태로 먼저 재생을 걸어두고, 사용자의 첫 클릭/터치/키 입력이 들어오는
// 즉시 음소거를 풀어 끊김 없이 소리가 나오도록 함 (브라우저의 자동재생 정책 우회)
function initBgm() {
    tryPlayCurrentBgm();
    const unlockAudio = () => {
        audioUnlocked = true;
        tryPlayCurrentBgm();
        document.removeEventListener('pointerdown', unlockAudio);
        document.removeEventListener('keydown', unlockAudio);
        document.removeEventListener('touchstart', unlockAudio);
    };
    document.addEventListener('pointerdown', unlockAudio);
    document.addEventListener('keydown', unlockAudio);
    document.addEventListener('touchstart', unlockAudio);
}
window.addEventListener('load', initBgm);

[1-E] JS - 전체화면 on/off (그대로 사용 가능, 게임 구조와 무관한 범용 코드)
=====================================================================

// iOS Safari는 일반 요소의 전체화면 API 자체를 지원하지 않아 아래 폴백으로도 해결되지 않음.
// 구형 안드로이드 브라우저/인앱 웹뷰 대응용 webkit 접두사 폴백만 추가.
function getFullscreenElement() {
    return document.fullscreenElement || document.webkitFullscreenElement;
}
function toggleFullscreen() {
    const el = document.documentElement;
    if (!getFullscreenElement()) {
        const request = el.requestFullscreen || el.webkitRequestFullscreen;
        if (request) {
            const result = request.call(el);
            if (result && result.catch) result.catch(() => {});
        }
    } else {
        const exit = document.exitFullscreen || document.webkitExitFullscreen;
        if (exit) exit.call(document);
    }
}
function updateFullscreenIcon() {
    const icon = document.getElementById('fullscreen-toggle-icon');
    if (icon) icon.className = getFullscreenElement() ? 'fa-solid fa-compress' : 'fa-solid fa-expand';
}
document.addEventListener('fullscreenchange', updateFullscreenIcon);
document.addEventListener('webkitfullscreenchange', updateFullscreenIcon);


=====================================================================
[2] JS - 화면 이동(홈/뒤로/앞으로) 로직 - ★ 게임마다 반드시 다시 짜야 함 ★
=====================================================================
아래는 "Pick the Word" 게임 기준 예시입니다. 이 게임은 화면이
intro-screen → category-screen → subcategory-screen → difficulty-screen → game-wrapper
순서로 진행되고, 각 화면 div가 class="hidden" 으로 켜짐/꺼짐 되는 구조입니다.

다른 게임은 화면 개수·이름·전환 함수 이름이 다르니, 아래 3가지를 그 게임에 맞게 바꿔서 쓰세요.
  1) NAV_SCREEN_IDS: 그 게임에 실제로 존재하는 화면 div id들을 화면 순서대로 나열
  2) navBack() / navForward(): 각 화면에서 뒤로/앞으로 눌렀을 때 실제로 호출할,
     그 게임에 이미 있는 화면 전환 함수 이름으로 교체
  3) navHome(): 처음 화면으로 되돌리는 그 게임의 "다시 시작" 함수 이름으로 교체

/* ================= 상단 네비게이션(홈/뒤로/앞으로) - 전체 화면 공통 ================= */
const NAV_SCREEN_IDS = ['intro-screen', 'category-screen', 'subcategory-screen', 'difficulty-screen', 'game-wrapper'];
// ↑ [다시 짜기] 이 게임에 실제로 있는 화면 id로 교체

function getCurrentScreenId() {
    for (const id of NAV_SCREEN_IDS) {
        const el = document.getElementById(id);
        if (el && !el.classList.contains('hidden')) return id;
    }
    return 'intro-screen'; // ↑ [다시 짜기] 이 게임의 첫 화면 id로 교체
}

function navHome() {
    if (getCurrentScreenId() === 'intro-screen') return; // ↑ [다시 짜기] 첫 화면 id
    restartGame(); // ↑ [다시 짜기] 이 게임의 "처음으로" 함수 이름
    updateNavButtonStates();
}

function navBack() {
    switch (getCurrentScreenId()) {
        case 'category-screen': backToIntroScreen(); break;
        case 'subcategory-screen': backToCategoryScreen(); break;
        case 'difficulty-screen': backToSubcategoryScreen(); break;
        case 'game-wrapper': exitGameToDifficulty(); break;
        default: break; // 첫 화면에서는 더 갈 곳이 없음
    }
    // ↑ [다시 짜기] case를 이 게임의 화면 id로, 오른쪽 함수를 이 게임의 "뒤로" 함수로 교체
    updateNavButtonStates();
}

function navForward() {
    switch (getCurrentScreenId()) {
        case 'intro-screen': showCategoryScreen(); break;
        case 'category-screen': forwardToSubcategoryScreen(); break;
        case 'subcategory-screen': confirmSubcategories(); break;
        case 'difficulty-screen': startSelectedGame(); break;
        default: break; // 게임 진행 화면에는 앞으로 갈 다음 단계가 없음
    }
    // ↑ [다시 짜기] case를 이 게임의 화면 id로, 오른쪽 함수를 이 게임의 "다음 단계로" 함수로 교체
    updateNavButtonStates();
}

// 화면이 바뀔 때마다(어떤 경로로 바뀌었든, 카드 클릭이든 버튼이든) 뒤로/앞으로 버튼의
// 활성/비활성 상태를 자동으로 갱신한다 - 이 부분은 그대로 써도 됨
function updateNavButtonStates() {
    const screen = getCurrentScreenId();
    const isFirst = screen === 'intro-screen'; // ↑ [다시 짜기] 첫 화면 id
    const isLast = screen === 'game-wrapper';  // ↑ [다시 짜기] "앞으로"가 의미 없는 마지막 화면 id
    const backBtn = document.getElementById('nav-back-btn');
    const homeBtn = document.getElementById('nav-home-btn');
    const fwdBtn = document.getElementById('nav-forward-btn');
    [backBtn, homeBtn].forEach(btn => {
        if (!btn) return;
        btn.disabled = isFirst;
        btn.style.opacity = isFirst ? '0.35' : '1';
        btn.style.pointerEvents = isFirst ? 'none' : '';
    });
    if (fwdBtn) {
        fwdBtn.disabled = isLast;
        fwdBtn.style.opacity = isLast ? '0.35' : '1';
        fwdBtn.style.pointerEvents = isLast ? 'none' : '';
    }
}

// 화면 전환이 직접 클릭(예: 카테고리 카드 클릭) 등 navForward/navBack을 거치지 않고도
// 일어날 수 있으므로, 각 화면 div의 class 변화를 감시해서 자동으로 버튼 상태를 갱신한다.
NAV_SCREEN_IDS.forEach(id => {
    const el = document.getElementById(id);
    if (el) new MutationObserver(updateNavButtonStates).observe(el, { attributes: true, attributeFilter: ['class'] });
});
updateNavButtonStates(); // 페이지 로드 시 1회 초기화

[3] 새 게임 파일에 붙일 때 체크리스트
=====================================================================
[ ] FontAwesome 6 CDN 링크가 <head>에 있는지 확인 (fa-solid, fa-regular 둘 다 필요)
[ ] [1-A] HTML을 <body> 맨 앞, 배경음악 <audio> 태그 바로 아래 붙여넣기
[ ] [1-B] CSS를 <style> 안에 붙여넣기 (이미 같은 이름의 클래스/규칙이 있으면 충돌 확인)
[ ] [1-C]~[1-E] JS를 <script> 안에 붙여넣기
      - playSound(type) 맨 앞에 `if (sfxMuted) return;` 추가하는 것 잊지 말기
      - <audio id="bgm-intro">, <audio id="bgm-game"> 태그가 실제로 있는지, 화면 전환할 때
        playBgm('intro') / playBgm('game') 을 호출해주고 있는지 확인
[ ] [2] 화면 이동 로직은 그 게임의 실제 화면 구조에 맞게 처음부터 다시 짜기
      - openHowToPlay() 함수 이름도 그 게임에 이미 있는 "게임방법 팝업 열기" 함수 이름으로
        onclick="openHowToPlay()" 부분을 바꿔줘야 함 (없으면 새로 만들어야 함)
[ ] 실제로 열어서 확인:
      - 첫 화면에서 홈/뒤로 버튼이 흐리게 비활성화되어 있는지
      - 마지막(게임) 화면에서 앞으로 버튼이 흐리게 비활성화되어 있는지
      - 효과음/배경음/전체화면 버튼 아이콘이 클릭할 때마다 바뀌는지
      - 창 크기를 줄였을 때 이 버튼들과 게임 화면이 겹치지 않는지
        (겹치면 게임 화면 배율 계산에 상단 버튼 높이만큼 여백을 빼줘야 함 - 대략 30px 정도)
